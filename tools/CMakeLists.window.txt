find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED exiv2 gexiv2)

set(ARGS)
list(APPEND DEPS_INCLUDE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/../lib ${CMAKE_BINARY_DIR}/lib/include)
list(APPEND DEPS_LIBRARIES jxl jxl_dec jxl_threads jxl_cms Comdlg32 Ole32)

if (JXLL_BUILD_WITH_WIC)
	list(APPEND DEPS_LIBRARIES Shlwapi windowscodecs)
    list(APPEND ARGS JXLL_BUILD_WITH_WIC)
elseif (JXLL_BUILD_WITH_FREEIMAGE)
    find_package(FreeImage QUIET)
    if (FreeImage_FOUND)
        list(APPEND ARGS JXLL_BUILD_WITH_FREEIMAGE)
        list(APPEND DEPS_INCLUDE_DIRS ${FreeImage_INCLUDE_DIRS})
        list(APPEND DEPS_LIBRARIES ${FreeImage_LIBRARIES})
        message(STATUS "Enable FreeImage for image I/O")
    endif()
elseif (JXLL_BUILD_WITH_IMAGEMAGICK)
    pkg_check_modules(IMAGEMAGICK IMPORTED_TARGET magick++)
    if (IMAGEMAGICK_FOUND)
        list(APPEND ARGS JXLL_BUILD_WITH_IMAGEMAGICK)
        list(APPEND DEPS_INCLUDE_DIRS ${IMAGEMAGICK_INCLUDE_DIRS})
        list(APPEND DEPS_LIBRARIES PkgConfig::IMAGEMAGICK)
        message(STATUS "Enable ImageMagick for image I/O")
    endif()
else()
	list(APPEND ARGS JXLL_BUILD_WITH_STB)
endif()

if (JXLL_BUILD_WITH_GUI)
    pkg_check_modules(GTKMM IMPORTED_TARGET gtkmm-4.0)
    if (GTKMM_FOUND)
        list(APPEND DEPS_INCLUDE_DIRS ${GTKMM_INCLUDE_DIRS})
        list(APPEND DEPS_LIBRARIES PkgConfig::GTKMM)
        message(STATUS "building gui tool with gtkmm")
        add_executable(gui gui.gtk.cc)
        target_compile_options(gui PRIVATE -msse4.2 -mpclmul -mavx2)
        target_compile_definitions(gui PRIVATE ${ARGS})
        target_include_directories(gui PRIVATE ${DEPS_INCLUDE_DIRS})
        target_link_libraries(gui PRIVATE ${DEPS_LIBRARIES})
    endif()
endif()

#
# CLI
add_executable(cli cli.cc tinyfiledialogs.c)
target_compile_options(cli PRIVATE -msse4.2 -mpclmul -mavx2)
target_compile_definitions(cli PRIVATE ${ARGS})
target_include_directories(cli PRIVATE ${DEPS_INCLUDE_DIRS})
target_link_libraries(cli PRIVATE ${DEPS_LIBRARIES})
